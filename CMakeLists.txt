cmake_minimum_required(VERSION 3.26)
if(POLICY CMP0207)
    cmake_policy(SET CMP0207 NEW)
endif()
project(RenderKit VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(RENDERKIT_REQUIRE_NATIVE_DEPENDENCIES "Require all dependencies for the native application" OFF)
if(RENDERKIT_REQUIRE_NATIVE_DEPENDENCIES)
    set(RENDERKIT_FIND_MODE REQUIRED)
else()
    set(RENDERKIT_FIND_MODE QUIET)
endif()

# The dependency-free foundation can be built while the native port is in progress.
add_library(renderkit_core STATIC src/core/config.cpp src/core/sequence.cpp src/core/sequence_names.cpp src/io/file_utils.cpp)
target_include_directories(renderkit_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
find_package(spdlog CONFIG ${RENDERKIT_FIND_MODE})
if(spdlog_FOUND)
    target_sources(renderkit_core PRIVATE src/core/logging.cpp)
    target_link_libraries(renderkit_core PUBLIC spdlog::spdlog)
endif()
find_package(OpenImageIO CONFIG ${RENDERKIT_FIND_MODE})
if(RENDERKIT_REQUIRE_NATIVE_DEPENDENCIES)
    find_package(OpenColorIO CONFIG REQUIRED)
endif()
if(OpenImageIO_FOUND)
    target_sources(renderkit_core PRIVATE src/core/sequence_fps.cpp src/io/oiio_cache.cpp src/io/image_reader.cpp src/processing/scaler.cpp src/processing/color_space.cpp src/processing/burnin.cpp src/processing/contact_sheet.cpp src/processing/frame_pipeline.cpp)
    target_link_libraries(renderkit_core PUBLIC OpenImageIO::OpenImageIO)
endif()

find_package(Qt6 COMPONENTS Widgets Svg ${RENDERKIT_FIND_MODE})
if(Qt6_FOUND)
    add_library(renderkit_ui STATIC src/ui/icon_manager.cpp src/ui/collapsible_group.cpp src/ui/preview_widget.cpp src/ui/splash_screen.cpp resources/renderkit.qrc)
    set_target_properties(renderkit_ui PROPERTIES AUTOMOC ON AUTORCC ON)
    target_include_directories(renderkit_ui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_link_libraries(renderkit_ui PUBLIC renderkit_core Qt6::Widgets Qt6::Svg)
    if(OpenImageIO_FOUND)
        target_sources(renderkit_ui PRIVATE src/ui/preview_image.cpp src/ui/workers.cpp src/ui/workers.hpp src/ui/main_window.cpp src/ui/main_window.hpp)
        target_compile_definitions(renderkit_ui PUBLIC RENDERKIT_HAS_OIIO=1)
        target_sources(renderkit_core PRIVATE src/processing/video_encoder.cpp src/core/converter.cpp)
        target_link_libraries(renderkit_core PUBLIC Qt6::Core)
        add_executable(RenderKit WIN32 src/main.cpp)
        target_link_libraries(RenderKit PRIVATE renderkit_ui)
        add_custom_command(TARGET RenderKit POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/resources/ocio" "$<TARGET_FILE_DIR:RenderKit>/data/ocio")
        if(WIN32 AND TARGET Qt6::QWindowsIntegrationPlugin)
            add_custom_command(TARGET RenderKit POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:RenderKit>/platforms"
                COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt6::QWindowsIntegrationPlugin>" "$<TARGET_FILE_DIR:RenderKit>/platforms")
        endif()
        if(WIN32 AND TARGET Qt6::QJpegPlugin)
            add_custom_command(TARGET RenderKit POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:RenderKit>/imageformats"
                COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt6::QJpegPlugin>" "$<TARGET_FILE_DIR:RenderKit>/imageformats")
        endif()
        if(WIN32 AND TARGET Qt6::QSvgIconPlugin)
            add_custom_command(TARGET RenderKit POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:RenderKit>/iconengines"
                COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt6::QSvgIconPlugin>" "$<TARGET_FILE_DIR:RenderKit>/iconengines")
        endif()
        if(WIN32 AND TARGET Qt6::QSvgPlugin)
            add_custom_command(TARGET RenderKit POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:RenderKit>/imageformats"
                COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt6::QSvgPlugin>" "$<TARGET_FILE_DIR:RenderKit>/imageformats")
        endif()
        if(WIN32)
            set(RENDERKIT_FFMPEG_PLATFORM windows)
        else()
            set(RENDERKIT_FFMPEG_PLATFORM linux)
        endif()
        set(RENDERKIT_FFMPEG_DIR "${CMAKE_SOURCE_DIR}/vendor/ffmpeg/${RENDERKIT_FFMPEG_PLATFORM}"
            CACHE PATH "Directory containing the portable FFmpeg bundle")
        if(EXISTS "${RENDERKIT_FFMPEG_DIR}")
            add_custom_command(TARGET RenderKit POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_directory "${RENDERKIT_FFMPEG_DIR}" "$<TARGET_FILE_DIR:RenderKit>/ffmpeg")
        endif()
        if(WIN32)
            install(TARGETS RenderKit RUNTIME_DEPENDENCY_SET renderkit_runtime RUNTIME DESTINATION .)
            foreach(plugin IN ITEMS QWindowsIntegrationPlugin QJpegPlugin QSvgPlugin QSvgIconPlugin)
                if(TARGET Qt6::${plugin})
                    if(plugin STREQUAL "QWindowsIntegrationPlugin")
                        set(plugin_dir platforms)
                    elseif(plugin STREQUAL "QSvgIconPlugin")
                        set(plugin_dir iconengines)
                    else()
                        set(plugin_dir imageformats)
                    endif()
                    install(FILES "$<TARGET_FILE:Qt6::${plugin}>" DESTINATION ${plugin_dir})
                endif()
            endforeach()
            install(RUNTIME_DEPENDENCY_SET renderkit_runtime
                DIRECTORIES "$<TARGET_FILE_DIR:RenderKit>"
                PRE_EXCLUDE_REGEXES "^[Aa][Pp][Ii]-[Mm][Ss]-" "^[Ee][Xx][Tt]-[Mm][Ss]-"
                POST_EXCLUDE_REGEXES "[/\\\\][Ww][Ii][Nn][Dd][Oo][Ww][Ss][/\\\\][Ss][Yy][Ss][Tt][Ee][Mm]32[/\\\\].*"
                RUNTIME DESTINATION .)
            install(DIRECTORY "${CMAKE_SOURCE_DIR}/resources/ocio" DESTINATION data)
            if(EXISTS "${RENDERKIT_FFMPEG_DIR}")
                install(DIRECTORY "${RENDERKIT_FFMPEG_DIR}/" DESTINATION ffmpeg)
            endif()
        elseif(UNIX AND NOT APPLE)
            set_target_properties(RenderKit PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
            install(TARGETS RenderKit RUNTIME_DEPENDENCY_SET renderkit_runtime RUNTIME DESTINATION .)
            foreach(plugin IN ITEMS QXcbIntegrationPlugin QOffscreenIntegrationPlugin QJpegPlugin QSvgPlugin QSvgIconPlugin)
                if(TARGET Qt6::${plugin})
                    if(plugin STREQUAL "QXcbIntegrationPlugin" OR plugin STREQUAL "QOffscreenIntegrationPlugin")
                        set(plugin_dir platforms)
                    elseif(plugin STREQUAL "QSvgIconPlugin")
                        set(plugin_dir iconengines)
                    else()
                        set(plugin_dir imageformats)
                    endif()
                    install(IMPORTED_RUNTIME_ARTIFACTS Qt6::${plugin}
                        RUNTIME_DEPENDENCY_SET renderkit_runtime LIBRARY DESTINATION Qt6/plugins/${plugin_dir})
                endif()
            endforeach()
            file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/install-config/qt.conf" CONTENT "[Paths]\nPlugins=Qt6/plugins\n")
            install(FILES "${CMAKE_CURRENT_BINARY_DIR}/install-config/qt.conf" DESTINATION .)
            install(PROGRAMS "${CMAKE_SOURCE_DIR}/scripts/renderkit-linux.sh" DESTINATION . RENAME renderkit)
            install(RUNTIME_DEPENDENCY_SET renderkit_runtime
                DIRECTORIES "$<TARGET_FILE_DIR:Qt6::Core>" "$<TARGET_FILE_DIR:OpenImageIO::OpenImageIO>"
                POST_EXCLUDE_REGEXES "^/(usr/)?lib.*/(ld-linux|lib(c|m|pthread|dl|rt|resolv)\\.so).*"
                LIBRARY DESTINATION lib)
            install(DIRECTORY "${CMAKE_SOURCE_DIR}/resources/ocio" DESTINATION data)
            if(EXISTS "${RENDERKIT_FFMPEG_DIR}")
                install(DIRECTORY "${RENDERKIT_FFMPEG_DIR}/" DESTINATION ffmpeg USE_SOURCE_PERMISSIONS)
            endif()
        endif()
    endif()
endif()
include(CTest)

    find_package(GTest CONFIG ${RENDERKIT_FIND_MODE})
    add_executable(test_native_foundation tests/cpp/test_foundation.cpp)
    target_link_libraries(test_native_foundation PRIVATE renderkit_core)
    add_test(NAME native_foundation COMMAND test_native_foundation)
    if(GTest_FOUND)
        add_executable(test_native_gtest tests/cpp/test_sequence_gtest.cpp)
        target_link_libraries(test_native_gtest PRIVATE renderkit_core GTest::gtest_main)
        add_test(NAME native_gtest COMMAND test_native_gtest)
    endif()
    if(spdlog_FOUND)
        add_executable(test_native_logging tests/cpp/test_logging.cpp)
        target_link_libraries(test_native_logging PRIVATE renderkit_core)
        add_test(NAME native_logging COMMAND test_native_logging)
        if(WIN32)
            set_tests_properties(native_logging PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:spdlog::spdlog>")
        endif()
    endif()
    if(OpenImageIO_FOUND)
        add_executable(test_native_cache tests/cpp/test_oiio_cache.cpp)
        target_link_libraries(test_native_cache PRIVATE renderkit_core)
        target_compile_definitions(test_native_cache PRIVATE RENDERKIT_TEST_OCIO_PATH="${CMAKE_CURRENT_SOURCE_DIR}/resources/ocio/config.ocio")
        add_test(NAME native_cache COMMAND test_native_cache)
        if(WIN32)
            set_tests_properties(native_cache PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:OpenImageIO::OpenImageIO>")
        endif()
    endif()
    if(TARGET renderkit_ui)
        find_package(Qt6Test CONFIG ${RENDERKIT_FIND_MODE})
        add_executable(test_native_ui tests/cpp/test_ui_foundation.cpp)
        target_link_libraries(test_native_ui PRIVATE renderkit_ui)
        add_test(NAME native_ui_foundation COMMAND test_native_ui)
        if(WIN32)
            set(RENDERKIT_TEST_PLATFORM minimal)
            set(RENDERKIT_QT_TEST_ENV "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>;QT_PLUGIN_PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>/../Qt6/plugins;QT_PLUGIN_PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>/../plugins")
        else()
            set(RENDERKIT_TEST_PLATFORM offscreen)
        endif()
        if(TARGET Qt6::Test)
            add_executable(test_native_qt tests/cpp/test_qt.cpp)
            set_target_properties(test_native_qt PROPERTIES AUTOMOC ON)
            target_link_libraries(test_native_qt PRIVATE renderkit_ui Qt6::Test)
            add_test(NAME native_qt COMMAND test_native_qt)
            set_tests_properties(native_qt PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${RENDERKIT_TEST_PLATFORM}")
            if(WIN32)
                set_tests_properties(native_qt PROPERTIES ENVIRONMENT_MODIFICATION "${RENDERKIT_QT_TEST_ENV}")
            endif()
        endif()
        set_tests_properties(native_ui_foundation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${RENDERKIT_TEST_PLATFORM}")
        if(WIN32)
            set_tests_properties(native_ui_foundation PROPERTIES ENVIRONMENT_MODIFICATION "${RENDERKIT_QT_TEST_ENV}")
        endif()
        if(OpenImageIO_FOUND)
            add_executable(test_native_encoder tests/cpp/test_video_encoder.cpp)
            target_link_libraries(test_native_encoder PRIVATE renderkit_core)
            if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ffmpeg/windows/ffmpeg.exe")
                target_compile_definitions(test_native_encoder PRIVATE RENDERKIT_TEST_FFMPEG="${CMAKE_CURRENT_SOURCE_DIR}/vendor/ffmpeg/windows/ffmpeg.exe")
            elseif(UNIX AND NOT APPLE AND EXISTS "${RENDERKIT_FFMPEG_DIR}/ffmpeg")
                target_compile_definitions(test_native_encoder PRIVATE RENDERKIT_TEST_FFMPEG="${RENDERKIT_FFMPEG_DIR}/ffmpeg")
            endif()
            add_test(NAME native_encoder COMMAND test_native_encoder)
            set_tests_properties(native_encoder PROPERTIES SKIP_RETURN_CODE 77)
            add_executable(test_native_window tests/cpp/test_main_window.cpp)
            target_link_libraries(test_native_window PRIVATE renderkit_ui)
            add_test(NAME native_window COMMAND test_native_window)
            set_tests_properties(native_window PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${RENDERKIT_TEST_PLATFORM}")
            if(WIN32)
                set_tests_properties(native_encoder PROPERTIES ENVIRONMENT_MODIFICATION "${RENDERKIT_QT_TEST_ENV}")
                set_tests_properties(native_window PROPERTIES ENVIRONMENT_MODIFICATION "${RENDERKIT_QT_TEST_ENV}")
            endif()
        endif()
    endif()
endif()
